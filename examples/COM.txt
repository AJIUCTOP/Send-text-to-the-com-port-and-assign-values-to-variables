int a=0;
int b=0;
int c=0;

void setup(){
pinMode(LED_BUILTIN, OUTPUT);
Serial.begin(115200);
}
void loop() {
if (Serial.available() > 0){
String data = Serial.readStringUntil('\n');
int R, G, B; //
sscanf (data.c_str(), "R%dG%dB%d", &R, &G, &B);
int a = R;
int b = G;
int c = B;
Serial.print(a);
Serial.print(' ');
Serial.print(b);
Serial.print(' ');
Serial.print(c);
Serial.print(' ');
if(c==1){
digitalWrite(LED_BUILTIN, HIGH);
delay(1000);                     
digitalWrite(LED_BUILTIN, LOW); 
delay(1000); }                 
}

}
